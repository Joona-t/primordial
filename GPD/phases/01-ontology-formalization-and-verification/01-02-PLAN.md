    ```python
    class AbsenceStateMachine(RuleBasedStateMachine):
        """Models an entity whose absence state evolves through transitions."""

        def __init__(self):
            super().__init__()
            self.state = None  # Set by @initialize
            self.transition_log = []

        @initialize(target=None)  # No bundle needed
        def init_state(self):
            """Start in any valid absence state."""
            # Pick a random initial state
            self.state = ... # drawn from V1_ABSENCE_STATES

        # One @rule per legal transition (or use a generic transition rule)
        @rule(target_state=st.sampled_from(sorted(V1_ABSENCE_STATES)))
        def attempt_transition(self, target_state):
            """Attempt a transition and verify it agrees with validate_transition()."""
            is_legal = validate_transition(self.state, target_state)
            if is_legal:
                self.transition_log.append((self.state, target_state))
                self.state = target_state

        @invariant()
        def state_is_valid(self):
            """Current state is always in V1_ABSENCE_STATES."""
            assert self.state in V1_ABSENCE_STATES

        @invariant()
        def no_illegal_reached(self):
            """No illegal transition was accepted."""
            for from_s, to_s in self.transition_log:
                assert validate_transition(from_s, to_s), \
                    f"Logged transition {from_s} -> {to_s} but validate_transition returns False"

        @invariant()
        def terminal_state_respected(self):
            """If we are in deleted, we can only stay in deleted."""
            if self.state == "deleted":
                for s in V1_ABSENCE_STATES:
                    if s != "deleted":
                        assert not validate_transition("deleted", s), \
                            f"deleted -> {s} should be illegal"

        @invariant()
        def initial_states_unreachable(self):
            """Cannot transition INTO not_invoked or not_generated (except self)."""
            for from_s, to_s in self.transition_log:
                if to_s in ("not_invoked", "not_generated") and from_s != to_s:
                    raise AssertionError(f"Reached initial state {to_s} from {from_s}")
    ```

    ```python
    # Build the illegal transition list from the transition table
    ILLEGAL_TRANSITIONS = [

    def test_illegal_transition_rejected(from_state, to_state):
        """Every illegal transition must be rejected by validate_transition()."""
        assert not validate_transition(from_state, to_state), \
            f"Expected illegal: {from_state} -> {to_state}"
    ```

    CATEGORY C: Transition table structural tests

    ```python
    def test_transition_table_completeness():
        """Every (source, target) pair in the 8-state product is classified."""
        classified = 0
        for from_s in V1_ABSENCE_STATES:
            for to_s in V1_ABSENCE_STATES:
                # validate_transition returns True or False, never raises for valid states
                result = validate_transition(from_s, to_s)
                assert isinstance(result, bool)
                classified += 1
        assert classified == 64

    def test_deleted_is_terminal():
        """deleted has no outgoing transitions except self."""
        for to_s in V1_ABSENCE_STATES:
            if to_s != "deleted":
                assert not validate_transition("deleted", to_s)

    def test_initial_states_unreachable():
        """not_invoked and not_generated have no incoming transitions except self."""
        for from_s in V1_ABSENCE_STATES:
            if from_s != "not_invoked":
                assert not validate_transition(from_s, "not_invoked")
            if from_s != "not_generated":
                assert not validate_transition(from_s, "not_generated")

    def test_self_transitions_legal():
        """Every state can self-transition (idempotent re-assignment)."""
        for state in V1_ABSENCE_STATES:
            assert validate_transition(state, state)

    def test_invalid_state_raises():
        """validate_transition rejects states not in V1_ABSENCE_STATES."""
        with pytest.raises(ValueError):
            validate_transition("resolved", "unknown")
        with pytest.raises(ValueError):
            validate_transition("unknown", "fake_state")
    ```
